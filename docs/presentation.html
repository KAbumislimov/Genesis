[Unit] Description=Campus MP3 player
After=sound.target network.target

[Service]
Type=simple  User=client1  Group=audio
ExecStartPre=mkdir -p /run/campus-player &&
            chown client1:audio /run/campus-player
ExecStart=/usr/bin/mpv
  --no-video     # только аудио
  --idle=yes     # не выходить после трека
  --volume-max=160 # >100% для гимна
  --input-ipc-server=/run/campus-player/mpv.sock
  --audio-device=alsa/default
Restart=always  RestartSec=3
StartLimitIntervalSec=0 # перезапуск всегда

Алгоритм выбора файла
# 1. День недели
DAY=$(date +%u) # 1=Пн…7=Вс

# 2. Сб/Вс → папка 1
[ $DAY -gt 5 ] && DAY=1

# 3. Путь к файлу
FILE="/mnt/music/Media/
     $DAY/$SLOT.mp3"

# 4. flock — один запуск
flock -n /tmp/cron.lock
  campus-playerctl play $FILE

# 5. 3 попытки с sleep 2
for i in 1 2 3; do
  play && break; sleep 2
done

scrape_configs:
  - job_name: node
    static_configs:
      # server — через docker host network
      - targets: ['127.0.0.1:9100']
        labels: {nodename: campus-server}
      # audio client
      - targets: ['<client1-ip>:9100']
        labels: {nodename: client1}
      # secondary client
      - targets: ['<client2-ip>:9100']
        labels: {nodename: client2}
    relabel_configs:
      # instance = nodename (читаемо в Grafana)
      - source_labels: [nodename]
        target_label: instance
Почему relabel_configs важен
Без него instance = 192.168.x.x:9100 — непонятно.
С ним instance = client1 — Grafana переменная $node работает корректно и дашборды показывают читаемые имена.

def has_perm(perm):
  role = current_user.role
  if perm == 'play' and
    role != 'admin' and
    _silence_active():
    return False
  return perm in
    ROLE_PERMS[role]
